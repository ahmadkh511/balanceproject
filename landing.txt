(function(){
    'use strict';

    // شريط التنقل - تأثير التمرير
    var landingNav = document.getElementById('landingNav');
    window.addEventListener('scroll', function() {
        landingNav.classList.toggle('scrolled', window.scrollY > 30);
    });

    // القائمة المتنقلة
    var mobileToggle = document.getElementById('landingMobileToggle');
    var mobileMenu = document.getElementById('landingMobileMenu');
    mobileToggle.addEventListener('click', function() {
        this.classList.toggle('active');
        mobileMenu.classList.toggle('open');
    });
    window.closeLandingMobile = function() {
        mobileToggle.classList.remove('active');
        mobileMenu.classList.remove('open');
    };
    mobileMenu.querySelectorAll('a[href^="#"]').forEach(function(l) {
        l.addEventListener('click', window.closeLandingMobile);
    });

    // تأثيرات الظهور
    var revealEls = document.querySelectorAll('.landing-reveal');
    var revealObs = new IntersectionObserver(function(entries) {
        entries.forEach(function(e) {
            if (e.isIntersecting) e.target.classList.add('active');
        });
    }, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' });
    revealEls.forEach(function(el) { revealObs.observe(el); });

    // عدّادات الإحصائيات
    var statNums = document.querySelectorAll('.landing-stat-num[data-target]');
    var statsCounted = false;
    var statsObs = new IntersectionObserver(function(entries) {
        entries.forEach(function(e) {
            if (e.isIntersecting && !statsCounted) {
                statsCounted = true;
                statNums.forEach(function(el) {
                    var target = parseInt(el.getAttribute('data-target'));
                    var step = target / 125;
                    var cur = 0;
                    var t = setInterval(function() {
                        cur += step;
                        if (cur >= target) {
                            cur = target;
                            clearInterval(t);
                        }
                        el.textContent = target >= 1000 ? Math.floor(cur).toLocaleString('ar-SA') : Math.floor(cur);
                    }, 16);
                });
            }
        });
    }, { threshold: 0.5 });
    if (statNums.length) {
        statsObs.observe(statNums[0].closest('.landing-stats-grid') || statNums[0]);
    }

})();
</script>
